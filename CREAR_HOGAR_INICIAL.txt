CREAR EL PRIMER HOGAR Y ASIGNAR AL USUARIO ACTUAL

1. Abre Supabase > SQL Editor.
2. Sustituye SOLO el email y el nombre del hogar del bloque siguiente.
3. Ejecuta el bloque una sola vez. No reutilices el mismo usuario para crear otro hogar.
4. Cierra sesión y vuelve a entrar en Oikos.

do $$
declare
  target_user uuid;
  new_household uuid;
begin
  select id into target_user
  from auth.users
  where email = 'ermabe31@gmail.com';

  if target_user is null then
    raise exception 'No existe ningún usuario de Auth con ese email.';
  end if;

  if exists (select 1 from public.household_members where user_id = target_user) then
    raise exception 'Ese usuario ya pertenece a un hogar; no se ha modificado nada.';
  end if;

  insert into public.households (name)
  values ('CASA V&J')
  returning id into new_household;

  insert into public.household_members (household_id, user_id, display_name, role)
  values (new_household, target_user, 'Jaime', 'owner');

  insert into public.furniture_rooms (household_id, name)
  select new_household, name from unnest(array[
    'Salón', 'Cocina', 'Habitación', 'Baño principal',
    'Baño invitados', 'Despacho', 'Terrazas'
  ]) as name;

  insert into public.expense_types (household_id, name)
  select new_household, name from unnest(array[
    'Supermercado', 'Luz', 'Agua', 'Internet', 'Ocio'
  ]) as name;
end $$;

PARA INVITAR A JAIME/VANESA DESPUÉS

Primero crea su usuario en Supabase > Authentication > Users. Después, en SQL Editor, ejecuta este bloque sustituyendo los valores:

insert into public.household_members (household_id, user_id, display_name, role)
select hm.household_id, u.id, 'NOMBRE-DEL-MIEMBRO', 'member'
from public.household_members hm
cross join auth.users u
where hm.user_id = (select id from auth.users where email = 'EMAIL-DEL-PROPIETARIO@example.com')
  and u.email = 'EMAIL-DEL-NUEVO-MIEMBRO@example.com';
